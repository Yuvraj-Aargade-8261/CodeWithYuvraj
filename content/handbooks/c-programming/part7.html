7.1

Linked Lists — Singly, doubly, circular, all operations

Singly linked list

A linked list is a chain of nodes where each node contains data and a pointer to the next node. Unlike arrays, linked lists allow O(1) insertion/deletion at any position (if you have the pointer), but O(n) access (no random access).

Hindi mein samjhoLinked list nodes ki chain hai jahan har node me data aur next node ka pointer hota hai. Arrays se different, linked lists me O(1) insertion/deletion hota hai (agar pointer ho), par O(n) access (random access nahi).
💻 C — Singly linked list
typedef struct Node {
    int data;
    struct Node *next;
} Node;

/* Create a new node */
Node *node_new(int data) {
    Node *n = malloc(sizeof(Node));
    n->data = data;
    n->next = NULL;
    return n;
}

/* Insert at head — O(1) */
void push_front(Node **head, int data) {
    Node *n = node_new(data);
    n->next = *head;
    *head = n;
}

/* Insert at tail — O(n) */
void push_back(Node **head, int data) {
    Node *n = node_new(data);
    if (!*head) { *head = n; return; }
    Node *curr = *head;
    while (curr->next) curr = curr->next;
    curr->next = n;
}

/* Delete node with value — O(n) */
void delete_node(Node **head, int target) {
    if (!*head) return;
    if ((*head)->data == target) {
        Node *temp = *head;
        *head = (*head)->next;
        free(temp);
        return;
    }
    Node *curr = *head;
    while (curr->next && curr->next->data != target)
        curr = curr->next;
    if (curr->next) {
        Node *temp = curr->next;
        curr->next = temp->next;
        free(temp);
    }
}

/* Print & free all */
void print_list(Node *head) {
    while (head) {
        printf("%d → ", head->data);
        head = head->next;
    }
    printf("NULL\n");
}

void free_list(Node **head) {
    while (*head) {
        Node *temp = *head;
        *head = (*head)->next;
        free(temp);
    }
}

Doubly linked list

💻 C — Doubly linked list node
typedef struct DNode {
    int data;
    struct DNode *prev;
    struct DNode *next;
} DNode;

/* Advantages over singly:
   - O(1) delete if you have the node pointer
   - Can traverse backwards
   - Used in: LRU cache, browser history, undo/redo */

DNode *dnode_new(int data) {
    DNode *n = calloc(1, sizeof(DNode));
    n->data = data;
    return n;  /* prev & next = NULL from calloc */
}

/* Insert after a given node — O(1) */
void insert_after(DNode *node, int data) {
    DNode *n = dnode_new(data);
    n->next = node->next;
    n->prev = node;
    if (node->next) node->next->prev = n;
    node->next = n;
}

7.2

Stacks — LIFO, array & linked list implementations, applications

Stack = LIFO (Last In, First Out)

A stack has two operations: push (add to top) and pop (remove from top). Think of a stack of plates. Used for: function call stack, expression parsing, undo operations, DFS traversal, balanced parentheses checking.

Hindi mein samjhoStack me do operations hain: push (top pe add) aur pop (top se remove). Plates ke stack jaisa sochho. Function call stack, expression parsing, undo, DFS, aur parentheses checking me use hota hai.
💻 C — Stack (array-based)
#define MAX_SIZE 100

typedef struct {
    int data[MAX_SIZE];
    int top;
} Stack;

void stack_init(Stack *s) { s->top = -1; }
int  stack_empty(Stack *s) { return s->top == -1; }
int  stack_full(Stack *s)  { return s->top == MAX_SIZE - 1; }

void push(Stack *s, int val) {
    if (stack_full(s)) { printf("Overflow!\n"); return; }
    s->data[++s->top] = val;
}

int pop(Stack *s) {
    if (stack_empty(s)) { printf("Underflow!\n"); return -1; }
    return s->data[s->top--];
}

int peek(Stack *s) {
    if (stack_empty(s)) return -1;
    return s->data[s->top];
}

/* Usage */
Stack s;
stack_init(&s);
push(&s, 10);
push(&s, 20);
push(&s, 30);
printf("%d\n", pop(&s));   /* 30 (LIFO) */
printf("%d\n", peek(&s));  /* 20 */
7.3

Queues — FIFO, circular queue, priority queue concepts

Queue = FIFO (First In, First Out)

A queue has enqueue (add to rear) and dequeue (remove from front). Think of a line at a ticket counter. Used for: BFS traversal, task scheduling, print spooling, message buffers in OS.

Hindi mein samjhoQueue me enqueue (rear pe add) aur dequeue (front se remove). Ticket counter ki line jaisa sochho. BFS, task scheduling, print spooling, OS message buffers me use hota hai.
💻 C — Circular queue (array-based)
typedef struct {
    int data[MAX_SIZE];
    int front, rear, count;
} Queue;

void queue_init(Queue *q) { q->front = q->rear = q->count = 0; }
int  queue_empty(Queue *q) { return q->count == 0; }
int  queue_full(Queue *q)  { return q->count == MAX_SIZE; }

void enqueue(Queue *q, int val) {
    if (queue_full(q)) { printf("Full!\n"); return; }
    q->data[q->rear] = val;
    q->rear = (q->rear + 1) % MAX_SIZE;  /* Wrap around! */
    q->count++;
}

int dequeue(Queue *q) {
    if (queue_empty(q)) { printf("Empty!\n"); return -1; }
    int val = q->data[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    q->count--;
    return val;
}

/* Circular: front and rear wrap around the array
   [_, _, _, front, d, d, rear, _, _]
   After dequeue+enqueue: wraps to beginning */

7.4

Binary Trees & BST — Insert, search, traversals, delete

Binary Search Tree (BST)

A BST is a binary tree where: left child < parent < right child. This property enables O(log n) search, insert, delete on average. Worst case (skewed): O(n). Used in: databases (B-trees), file systems, symbol tables.

Hindi mein samjhoBST ek binary tree hai jahan: left child < parent < right child. Isse average me O(log n) search, insert, delete milta hai. Worst case (skewed): O(n). Databases, file systems, symbol tables me use hota hai.
💻 C — Binary Search Tree
typedef struct TreeNode {
    int data;
    struct TreeNode *left, *right;
} TreeNode;

TreeNode *tree_new(int data) {
    TreeNode *n = malloc(sizeof(TreeNode));
    n->data = data;
    n->left = n->right = NULL;
    return n;
}

/* Insert into BST — O(log n) avg */
TreeNode *bst_insert(TreeNode *root, int data) {
    if (!root) return tree_new(data);
    if (data < root->data)
        root->left = bst_insert(root->left, data);
    else if (data > root->data)
        root->right = bst_insert(root->right, data);
    return root;
}

/* Search in BST — O(log n) avg */
TreeNode *bst_search(TreeNode *root, int target) {
    if (!root || root->data == target) return root;
    if (target < root->data)
        return bst_search(root->left, target);
    return bst_search(root->right, target);
}

/* Inorder traversal — prints sorted! */
void inorder(TreeNode *root) {
    if (!root) return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

/* Build BST: 50, 30, 70, 20, 40, 60, 80 */
TreeNode *root = NULL;
int vals[] = {50, 30, 70, 20, 40, 60, 80};
for (int i = 0; i < 7; i++)
    root = bst_insert(root, vals[i]);
inorder(root);  /* 20 30 40 50 60 70 80 (sorted!) */

Inorder (LNR)Left → Node → RightSorted output from BST
Preorder (NLR)Node → Left → RightCopy tree, serialize
Postorder (LRN)Left → Right → NodeDelete tree, evaluate expressions

7.5

Hash Tables — Hash functions, collision handling, O(1) lookup

Hash table with chaining

A hash table maps keys to values using a hash function. Average case: O(1) for insert, search, delete. Collisions (two keys → same index) are handled by chaining (linked list at each bucket) or open addressing.

Hindi mein samjhoHash table hash function use karke keys ko values se map karta hai. Average case: O(1) insert, search, delete. Collision (do keys → same index) chaining (har bucket pe linked list) ya open addressing se handle hota hai.
💻 C — Hash table (string keys)
#define TABLE_SIZE 64

typedef struct Entry {
    char *key;
    int value;
    struct Entry *next;  /* Chaining */
} Entry;

typedef struct {
    Entry *buckets[TABLE_SIZE];
} HashTable;

/* djb2 hash function (classic) */
unsigned int hash(const char *key) {
    unsigned int h = 5381;
    while (*key) h = h * 33 + *key++;
    return h % TABLE_SIZE;
}

void ht_set(HashTable *ht, const char *key, int val) {
    unsigned int idx = hash(key);
    Entry *e = malloc(sizeof(Entry));
    e->key = strdup(key);
    e->value = val;
    e->next = ht->buckets[idx];  /* Prepend to chain */
    ht->buckets[idx] = e;
}

int ht_get(HashTable *ht, const char *key) {
    Entry *e = ht->buckets[hash(key)];
    while (e) {
        if (strcmp(e->key, key) == 0) return e->value;
        e = e->next;
    }
    return -1;  /* Not found */
}
7.6

Sorting Algorithms — Bubble, selection, merge sort, quicksort

Essential sorting algorithms

💻 C — Merge sort (O(n log n) guaranteed)
void merge(int *arr, int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int *L = malloc(n1 * sizeof(int));
    int *R = malloc(n2 * sizeof(int));
    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int i = 0; i < n2; i++) R[i] = arr[m + 1 + i];

    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2)
        arr[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
    free(L); free(R);
}

void merge_sort(int *arr, int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        merge_sort(arr, l, m);
        merge_sort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
}

Interview tip: Know when to use which: Quicksort for general purpose (fastest in practice), Merge sort when stability matters, Insertion sort for nearly-sorted small arrays, Heap sort when O(1) space is required.

Big-O Complexity — Time & space analysis, common patterns

Big-O cheat sheet

Big-O describes how runtime/space grows as input size grows. It ignores constants and lower-order terms. Knowing Big-O for every data structure operation is essential for technical interviews and system design.

Hindi mein samjhoBig-O batata hai input size badhne pe runtime/space kaise badhta hai. Constants aur lower-order terms ignore karta hai. Har data structure operation ka Big-O jaanna interviews aur system design ke liye zaroori hai.

💻 C — Recognizing complexity patterns
/* O(1) — constant */
arr[i];  hash_table[key];

/* O(log n) — halving each step */
while (n > 0) n /= 2;

/* O(n) — single loop */
for (int i = 0; i < n; i++) { /* ... */ }

/* O(n log n) — divide & conquer */
merge_sort(arr, 0, n - 1);

/* O(n²) — nested loops */
for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) { /* ... */ }
Data structure selection rule: Need fast lookup? → Hash table O(1). Need sorted data? → BST O(log n). Need fast insert/delete? → Linked list O(1). Need random access? → Array O(1). Need FIFO? → Queue. Need LIFO? → Stack.

Quicksort implementation

💻 C — Quicksort with Lomuto partition
void swap(int *a, int *b) {
    int t = *a; *a = *b; *b = t;
}

int partition(int *arr, int lo, int hi) {
    int pivot = arr[hi];  /* Last element as pivot */
    int i = lo - 1;
    for (int j = lo; j < hi; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[hi]);
    return i + 1;
}

void quicksort(int *arr, int lo, int hi) {
    if (lo < hi) {
        int p = partition(arr, lo, hi);
        quicksort(arr, lo, p - 1);
        quicksort(arr, p + 1, hi);
    }
}

/* Binary search — O(log n) */
int binary_search(int *arr, int n, int target) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;  /* Avoids overflow */
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;  /* Not found */
}
